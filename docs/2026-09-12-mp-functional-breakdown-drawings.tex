% Colours: light blue = tdns (as in dsync-drawings.tex), iis ocean-light =
% tdns-transport, iis lemon-light = tdns-mp.
%
% \input this file into a deck that already defines \prt and the iis
% colours; \providecolor and \providecommand below only fill in what is
% missing, so the standalone wrapper compiles too. Needs the tikz libraries
% positioning, fit and calc.
%

\providecolor{light blue}{rgb}{0.8,1,1}
\providecolor{iis ocean-light}{HTML}{a7d8fd}
\providecolor{iis lemon-light}{HTML}{ffe696}
\providecolor{iis concrete}{HTML}{d8d8d8}
\providecommand{\prt}[1]{\textbf{\texttt{#1}}}

\tikzset{
  mpblk/.style={draw=black, rectangle, text width=3.35cm, minimum height=9mm,
    align=center, line width=0.8pt, inner sep=2pt, font=\tiny},
  mpblk-mp/.style={mpblk, fill=iis lemon-light},
  mpblk-tp/.style={mpblk, fill=iis ocean-light},
  mpblk-td/.style={mpblk, fill=light blue},
  mpblk-other/.style={mpblk, fill=iis concrete, rounded corners=1pt},
  mpband/.style={draw=black, dashed, rounded corners=2pt, inner xsep=2.5mm, inner ysep=3mm},
  mpdep/.style={->, blue, line width=0.4mm},
  mpflow/.style={->, black, line width=0.4mm},
  mpwire/.style={->, blue, line width=0.4mm, dashed},
}

% Drawing 2: one application message, sender column and receiver column

\newcommand{\mpbreakdownmsgpath}{%
  \begin{tikzpicture}[x=1cm, y=1.1cm,
      pth/.style={mpblk, text width=5.3cm, minimum height=9mm}]

    % --- sender, top to bottom -------------------------------------------
    \node[pth, fill=iis lemon-light] (s1) at (0,1)
      {verb table builds the carrier\\\prt{mp\_send.go}: \prt{AppMessage\{Scope, TypeToken, Payload\}}};
    \node[pth, fill=iis ocean-light] (s2) at (0,0)
      {pick mechanism, fall back, retry until confirmed\\\prt{TransportManager.Send}, \prt{ReliableMessageQueue}};
    \node[pth, fill=iis ocean-light] (s3) at (0,-1)
      {build the NOTIFY, register the pending confirmation\\\prt{DNSTransport.sendNotifyWithPayload}};
    \node[pth, fill=iis ocean-light] (s4) at (0,-2)
      {encrypt for the peer, sign with our key\\\prt{SecurePayloadWrapper.WrapOutgoing} $\to$ \prt{crypto/jose}};
    \node[pth, fill=light blue] (s5) at (0,-3)
      {payload into the EDNS0 CHUNK option, Format = envelope\\\prt{edns0.CreateChunkOption}};
    \node[pth, fill=iis concrete, rounded corners=1pt] (s6) at (0,-4)
      {NOTIFY(CHUNK) on the wire\\\prt{dns.Client} to the peer's DNS endpoint};

    \foreach \a/\b in {s1/s2, s2/s3, s3/s4, s4/s5, s5/s6}
      \draw[mpflow] (\a.south) -- (\b.north);

    % --- receiver, bottom to top -----------------------------------------
    \node[pth, fill=light blue] (r1) at (7.8,-4)
      {tdns receives it, dispatches by qtype\\\prt{NotifyHandler} $\to$ \prt{RegisterNotifyHandler(CHUNK)}};
    \node[pth, fill=iis ocean-light] (r2) at (7.8,-3)
      {qname $\to$ distribution id and sender; envelope check\\\prt{ChunkNotifyHandler.RouteViaRouter}};
    \node[pth, fill=iis lemon-light] (r3) at (7.8,-2)
      {is this sender allowed? (config list or HSYNC3)\\\prt{IsPeerAuthorized}, before any crypto};
    \node[pth, fill=iis ocean-light] (r4) at (7.8,-1)
      {verify with the sender's key only, then decrypt\\\prt{UnwrapIncomingFromPeerEnvelope} $\to$ \prt{crypto/jose}};
    \node[pth, fill=iis lemon-light] (r5) at (7.8,0)
      {read verb, originator, zone, nonce\\\prt{parseAppPayload} $\to$ \prt{IncomingMessage}};
    \node[pth, fill=iis ocean-light] (r6) at (7.8,1)
      {middleware chain, then the verb's handler; inline confirmation\\\prt{DNSMessageRouter.Route} $\to$ \prt{handleApp<Verb>}};
    \node[pth, fill=iis lemon-light] (r7) at (7.8,2)
      {hand to the engine's queue\\\prt{RouteToCallback} $\to$ \prt{routeIncomingMessage} $\to$ \prt{MsgQs}};

    \foreach \a/\b in {r1/r2, r2/r3, r3/r4, r4/r5, r5/r6, r6/r7}
      \draw[mpflow] (\a.north) -- (\b.south);

    % --- the wire and the confirmation back ------------------------------
    \draw[mpwire] (s6.east) -- (r1.west)
      node[midway, above, font=\tiny] {NOTIFY(CHUNK)};
    \draw[mpwire] (r6.west) -- ++(-1.1,0) |- (s3.east)
      node[pos=0.5, left, font=\tiny, rotate=90, anchor=south, xshift=-0.55cm, yshift=0.1cm] {response: inline confirmation};

    % --- column titles and legend ----------------------------------------
    \node[anchor=south, font=\scriptsize\bfseries] at ($(s1.north) + (0,0.15)$) {sender};
    \node[anchor=south, font=\scriptsize\bfseries] at ($(r7.north) + (0,0.15)$) {receiver};

    \node[mpblk-mp, text width=1.7cm, minimum height=5mm] (lg1) at (0.9,-5.3) {tdns-mp};
    \node[mpblk-tp, text width=1.7cm, minimum height=5mm] (lg2) at (3.1,-5.3) {tdns-transport};
    \node[mpblk-td, text width=1.7cm, minimum height=5mm] (lg3) at (5.3,-5.3) {tdns};
    \node[mpblk-other, text width=1.7cm, minimum height=5mm] (lg4) at (7.5,-5.3) {the wire};
  \end{tikzpicture}%
}
